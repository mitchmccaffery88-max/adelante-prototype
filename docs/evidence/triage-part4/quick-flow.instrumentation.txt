async function installInstrumentation() {
 await page.evaluate(() => {
   const w=window as any; w.__triage = {events:[], pending: new Map(), rows:[], sections:0};
   const t=w.__triage;
   const sample=() => { const el=document.querySelector('[data-testid="quick-action-settings"]'); const rows=Array.from(el?.querySelectorAll('li[data-testid^="pref-"]')??[]).map(x=>x.getAttribute("data-testid")); const count=el?1:0; if(JSON.stringify(rows)!==JSON.stringify(t.rows)||count!==t.sections) {t.events.push({time:performance.now(),url:location.pathname,rows,section:count,type:count<t.sections?"unmounted":count>t.sections?"mounted":"rows changed"});t.rows=rows;t.sections=count;} };
   new MutationObserver(sample).observe(document.body,{subtree:true,childList:true,characterData:true}); sample();
   window.addEventListener("adelante:quick-actions-changed",()=>t.events.push({type:"prefs changed",time:performance.now(),url:location.pathname}));
   for(const [set,clear,repeat] of [["setTimeout","clearTimeout",false],["setInterval","clearInterval",true]] as const){const original=w[set].bind(w), cancel=w[clear].bind(w); w[set]=(fn:any,delay:any,...args:any[])=>{let id:any;id=original((...inner:any[])=>{if(!repeat)t.pending.delete(id);if(typeof fn==="function")fn(...inner);else (0,eval)(fn);},delay,...args);t.pending.set(id,{type:set,delay,created:performance.now(),stack:new Error().stack?.split("\n").slice(2,5)});return id;}; w[clear]=(id:any)=>{t.pending.delete(id);return cancel(id);};}
 });
}
const snapshots:any[]=[];
async function snapshot(label:string){
 for(const name of ["roles","quickActions","quickActionPrefs"]) if(!moduleUrls.has(name))throw new Error("Loaded module URL missing: "+name);
 const result=await page.evaluate(async (urls)=>{
  const roles=await import(/* @vite-ignore */ urls.roles); const qa=await import(/* @vite-ignore */ urls.quickActions); const prefs=await import(/* @vite-ignore */ urls.quickActionPrefs);
  const staff=roles.getActingStaff(); const actor={role:roles.getActingRole(),staffId:staff.id,staffName:staff.name,clinicianId:staff.clinicianId,caseManagerId:staff.caseManagerId};
  const t=(window as any).__triage;const section=document.querySelector('[data-testid="quick-action-settings"]');
  return {url:location.pathname,actor,preference:localStorage.getItem(prefs.preferenceKey(staff.id)),defaults:qa.quickActionsFor(actor).map((a:any)=>a.id),optionIds:prefs.personalActionOptions(staff).map((o:any)=>o.action.id),rows:Array.from(section?.querySelectorAll('li[data-testid^="pref-"]')??[]).map(x=>x.getAttribute("data-testid")),settingsMounted:!!section,events:t.events,pendingTimers:Array.from(t.pending.values()),timerScope:"Timers scheduled after initial dev-hook readiness; pre-existing timers not introspectable"};
 },Object.fromEntries(moduleUrls));
 snapshots.push({label,...result});writeFileSync(evidence+"/quick-flow-readback.json",JSON.stringify(snapshots,null,2));console.log(label,JSON.stringify(result));return result;
}
